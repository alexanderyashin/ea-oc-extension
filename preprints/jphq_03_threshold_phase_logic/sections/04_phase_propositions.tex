\section{Propositions on threshold-induced transitions}
\label{sec:props}

\begin{proposition}[Uniqueness of phase classification]
\label{prop:unique-phase}
For every admissible state $s \in \Omega(K_{EA})$, the phase classification
\[
\Pi(s) \in \{\texttt{SUCCESS}, \texttt{INERTIA}, \texttt{COLLAPSE}, \texttt{STOP}\}
\]
is uniquely determined by the ETS phase predicates together with the fixed
precedence ordering.
\end{proposition}

\begin{proof}[Sketch]
By Definition~\ref{sec:phases}, each phase predicate is Boolean-valued.
For any state $s$, either exactly one predicate is true or multiple predicates
are simultaneously true. In the latter case, the total precedence ordering
\[
\texttt{STOP} \succ \texttt{COLLAPSE} \succ \texttt{INERTIA} \succ \texttt{SUCCESS}
\]
selects a unique maximal element. Hence the classification map $\Pi$ is
well-defined and single-valued on $\Omega(K_{EA})$.
\end{proof}

\begin{proposition}[Discontinuous reassignment under threshold crossings]
\label{prop:discontinuous}
Let $s(\lambda)$ be a parametrized family of admissible states in
$\Omega(K_{EA})$, with $\lambda$ an arbitrary external parameter.
If, for some $\lambda^\ast$, the truth value of at least one ETS phase predicate
changes due to a threshold crossing (e.g.\ a sign change in $f_k(s(\lambda))$ or
loss of observability), then the induced phase classification
$\Pi(s(\lambda))$ may change discontinuously at $\lambda^\ast$.
\end{proposition}

\begin{proof}[Sketch]
Each phase predicate defines a subset of $\Omega(K_{EA})$ via strict Boolean
conditions. A change in predicate truth value corresponds to crossing the
boundary between disjoint predicate-defined subsets. Since ETS imposes no
continuity or smoothness requirements on the parametrization $\lambda$,
the induced map $\lambda \mapsto \Pi(s(\lambda))$ need not be continuous.
\end{proof}

\begin{proposition}[STOP as an absorbing phase]
\label{prop:stop-absorbing}
If $\Pi(s)=\texttt{STOP}$ for some $s \in \Omega(K_{EA})$, then no ETS-allowed
phase transition exists from $s$ to any other phase class.
\end{proposition}

\begin{proof}[Sketch]
By the ETS allowance relation,
\[
\texttt{STOP} \Rightarrow \varnothing .
\]
Therefore, there are no admissible outgoing phase-level transitions from STOP.
\end{proof}

\begin{proposition}[Irreversibility of \texttt{COLLAPSE} at the phase level]
\label{prop:collapse-no-return}
If $\Pi(s)=\texttt{COLLAPSE}$ for some $s \in \Omega(K_{EA})$, then no ETS-allowed
phase transition leads directly from COLLAPSE to either SUCCESS or INERTIA.
\end{proposition}

\begin{proof}[Sketch]
According to the ETS allowance relation,
\[
\texttt{COLLAPSE} \Rightarrow \{\texttt{STOP}\}.
\]
No other outgoing phase-level transitions from COLLAPSE are permitted.
\end{proof}

\begin{proposition}[Diagnostic non-steerability]
\label{prop:non-steerable}
Phase classification under ETS does not imply the existence of a continuous,
controllable, or reversible path between phase classes within
$\Omega(K_{EA})$.
\end{proposition}

\begin{proof}[Sketch]
ETS specifies phase transitions as a guarded allowance relation, not as a
dynamical system or control model. The existence of a phase label or an allowed
transition does not entail reachability, controllability, or reversibility of
state trajectories within $\Omega(K_{EA})$.
\end{proof}

\begin{remark}[Non-claims]
The propositions above introduce no optimization objectives, recovery strategies,
governance prescriptions, or continuity assumptions. They are logical
consequences of ETS predicates, precedence ordering, and allowance rules only.
\end{remark}
