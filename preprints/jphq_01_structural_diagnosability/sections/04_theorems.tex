% Formal properties of structural diagnosability D(K_EA).
% This section establishes:
%   (i) well-definedness under ETS-equivalent representations, and
%   (ii) decidability under the finiteness assumptions implicit in ETS.K_EA.v1.1.
%
% No new primitives are introduced. All arguments rely exclusively on
% OC Core semantics and ETS.K_EA.v1.1-conformant representations.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\label{sec:theorems}

\begin{theorem}[Well-definedness of structural diagnosability]
Let \KEA{} be an admissible enterprise-continuum instance. The Boolean property
\(D(\KEA)\) is well-defined in the following sense: if two concrete
representations \(R_1\) and \(R_2\) are equivalent under
\texttt{ETS.K\_EA.v1.1}, then they induce the same value of \(D(\KEA)\).
\end{theorem}

\begin{proof}[Proof sketch]
ETS-equivalent representations induce the same \OC{}-core objects:
\(\Om(\KEA)\), \(\dOm(\KEA)\), \(\Ax(\KEA)\), \(\Th(\KEA)\), \(\Flow(\KEA)\),
\(\Cyc(\KEA)\), and \(\kcont(\KEA)\).
Consequently, they induce the same finite encoding domain \(\mathsf{Enc}(\KEA)\)
and the same diagnostic predicate family \(\mathsf{Pred}_D(\KEA)\).

The value of \(D(\KEA)\) depends exclusively on whether all predicate evaluations
\(P(x)\), for \(P \in \mathsf{Pred}_D(\KEA)\) and \(x \in \mathsf{Enc}(\KEA)\),
are well-defined and computable under the fixed \OC{}/ETS semantics. Since both
the domain and semantics are invariant under ETS-equivalent representations, the
truth of this condition is invariant as well. Therefore \(D(\KEA)\) is
representation-independent.
\end{proof}

\begin{theorem}[Decidability of structural diagnosability]
Let an admissible \KEA{} instance be given by a finite
\texttt{ETS.K\_EA.v1.1}-conformant representation. Then the structural
diagnosability property \(D(\KEA)\) is decidable.
\end{theorem}

\begin{proof}[Proof sketch]
By the representational assumption, both the encoding set
\(\mathsf{Enc}(\KEA)\) and the predicate family \(\mathsf{Pred}_D(\KEA)\) are
finite.

Each predicate \(P \in \mathsf{Pred}_D(\KEA)\) is defined by explicit,
algorithmically specified \OC{}/ETS rules (membership in \(\Om\), boundary
checks for \(\dOm\), threshold satisfaction, and evaluation of
\(\kcont\) in the prescribed numeric domain). Therefore, for any fixed
\(x \in \mathsf{Enc}(\KEA)\), the computation of \(P(x)\) either terminates with
a Boolean value or fails to be well-defined.

To decide \(D(\KEA)\), one enumerates all pairs
\((P,x) \in \mathsf{Pred}_D(\KEA) \times \mathsf{Enc}(\KEA)\) and checks whether
each evaluation \(P(x)\) is well-defined and computable. Since this is a finite
set of checks over finite data, the procedure terminates. Consequently,
\(D(\KEA)\) is decidable, as a standard finite decision problem over a finite
encoding domain \cite{Sipser2012,Clarke2018}.
\end{proof}
