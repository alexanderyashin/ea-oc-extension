% ----------------------------------------------------------------------------
% 0) Small internal helpers (guarded)

% \AR@tt{<raw>} : render raw token string as typewriter, safe for underscores, etc.
\@ifundefined{AR@tt}{%
  \def\AR@tt#1{\texttt{\detokenize{#1}}}%
}{}

% \AR@ttm{<raw>} : same but safe in math mode too (forces text)
\@ifundefined{AR@ttm}{%
  \def\AR@ttm#1{%
    \ifmmode
      \text{\AR@tt{#1}}%
    \else
      \AR@tt{#1}%
    \fi
  }%
}{}

% \AR@ensuremath{<x>} : minimal ensure-math without importing extra packages
\@ifundefined{AR@ensuremath}{%
  \def\AR@ensuremath#1{\ifmmode #1\else $#1$\fi}%
}{}

% Helper: define a macro \X that is either sym or fun depending on next char
% Usage: \AR@def@maybe@arg{X}{<symbol>}{<function body using ##1>}
\@ifundefined{AR@def@maybe@arg}{%
  \def\AR@def@maybe@arg#1#2#3{%
    \expandafter\def\csname #1\endcsname{%
      \@ifnextchar\bgroup{\csname #1@with\endcsname}{#2}%
    }%
    \expandafter\def\csname #1@with\endcsname##1{#3}%
  }%
}{}

% Canonical OC tuple components (minimal names)
