% --- Pagination freeze (wrapper-only)
% Purpose: JPHQ sources may contain \pagenumbering{...} or \setcounter{page}{...}
% Policy: JPHQ is immutable; neutralize page resets at aggregation layer only.
\makeatletter
\newcommand{\ARFreezePaginationBegin}{%
  % save originals (idempotent per chapter)
  \let\AR@orig@pagenumbering\pagenumbering%
  \let\AR@orig@setcounter\setcounter%
  % neutralize \pagenumbering{...}
  \renewcommand{\pagenumbering}[1]{}%
  % neutralize \setcounter{page}{...} but keep all other counters functional
  \renewcommand{\setcounter}[2]{%
    \def\AR@ctr{##1}\def\AR@page{page}%
    \ifx\AR@ctr\AR@page
      % ignore
    \else
      \AR@orig@setcounter{##1}{##2}%
    \fi
  }%
}
\newcommand{\ARFreezePaginationEnd}{%
  \let\pagenumbering\AR@orig@pagenumbering%
  \let\setcounter\AR@orig@setcounter%
}
\makeatother
