% Purpose: Provide a minimal checklist to evaluate whether a prompt/output stays in OPS-2A scope.
% Constraints: diagnostic-only; checklist is not an action plan.

\section{Appendix: Minimal Conformance Checklist}

\begin{enumerate}[leftmargin=*, itemsep=0.35em]
  \item Is the output limited to (scope label, OK/STOP classification, minimal rationale)?
  \item Are there \emph{no} recommendations, remediations, or operational next steps?
  \item Is STOP treated as terminal (no ``if you do X, you can proceed'')?
  \item Is all provenance either provided or explicitly marked unknown (no inference)?
  \item Are any attempts at evidence inflation classified as misuse/STOP?
\end{enumerate}
